120template <class SimplexType, class LayoutStridedPolicy, class MemorySpace>
121class LocalChain
122{
123public:
124 using memory_space = MemorySpace;
125 using simplex_type = SimplexType;
126 using discrete_element_type = typename simplex_type::discrete_element_type;
127 using discrete_vector_type = typename simplex_type::discrete_vector_type;
128 static constexpr std::size_t MAX_SIZE
129 = 1UL << ddc::type_seq_size_v<ddc::to_type_seq_t<discrete_vector_type>>;
130 using storage_type = std::array<simplex_type, MAX_SIZE>;
131
132 using iterator_type = LocalChainIterator<SimplexType, LayoutStridedPolicy, MemorySpace>;
133 using const_iterator_type = iterator_type;
134
135private:
136 static constexpr bool s_is_local = true;
137 static constexpr std::size_t s_k = simplex_type::dimension();
138 discrete_element_type m_origin = misc::filled_struct<discrete_element_type>();
139 storage_type m_vects {};
140 std::size_t m_size = 0;
141
142public:
143 KOKKOS_DEFAULTED_FUNCTION constexpr LocalChain() = default;
144 KOKKOS_DEFAULTED_FUNCTION constexpr LocalChain(LocalChain const&) = default;
145 KOKKOS_DEFAULTED_FUNCTION constexpr LocalChain(LocalChain&&) = default;

163 template <misc::NotSpecialization<ddc::DiscreteVector>... T>
164 constexpr explicit LocalChain(discrete_element_type origin, T... simplex) noexcept
165 : m_origin(origin)
166 , m_size(sizeof...(T))
167 {
168 static_assert(sizeof...(T) <= MAX_SIZE);
169 std::size_t i = 0;
170 ((m_vects[i++] = simplex), ...);
171 assert(check() == 0 && "there are duplicate simplices in the chain");
172 if constexpr (sizeof...(T) > 1) {
173 assert(misc::are_all_equal(simplex.discrete_element()...)
174 && "LocalChain must contain simplices with same origin (if not, use Chain)");
175 }
176 assert((!simplex.negative() && ...)
177 && "negative simplices are not supported in LocalChain");
178 }

363 template <class T>
364 constexpr LocalChain& operator*=(T t)
365 {
366 if (t == 1) {
367 } else if (t == -1) {
368 for (std::size_t i = 0; i < m_size; ++i) {
369 m_vects[i] = -m_vects[i];
370 }
371 } else {
372 assert(false && "chain must be multiplied by 1 or -1");
373 }
374 return *this;
375 }

385 constexpr bool operator==(LocalChain const& simplices) const
386 {
387 if (m_size != simplices.m_size) {
388 return false;
389 }
390 auto simplex = begin();
391 auto other_simplex = simplices.begin();
392 for (; simplex < end(); ++simplex, ++other_simplex) {
393 if (*simplex != *other_simplex) {
394 return false;
395 }
396 }
397 return true;
398 }

403template <std::size_t K, misc::Specialization<ddc::DiscreteDomain> Dom>
404struct TangentBasis;
405
406template <std::size_t K, class... Tag>
407struct TangentBasis<K, ddc::DiscreteDomain<Tag...>>
408{
409 template <class MemorySpace = Kokkos::HostSpace, class Elem>
410 static constexpr auto operator()(Elem elem)
411 {
412 using chain_type = LocalChain<Simplex<K, Tag...>, Kokkos::LayoutRight, MemorySpace>;
413 std::array<std::ptrdiff_t, sizeof...(Tag)> permutation
414 = {0 * ddc::type_seq_rank_v<Tag, ddc::TypeSeq<Tag...>>...};
415 for (auto i = permutation.begin(); i < permutation.begin() + K; ++i) {
416 *i = 1;
417 }
418 chain_type basis {typename chain_type::discrete_element_type(elem)};
419 do {
420 typename chain_type::discrete_vector_type vect;
421 ddc::detail::array(vect) = permutation;
422 basis += vect;
423 } while (std::prev_permutation(permutation.begin(), permutation.end()));
424 return basis;
425 }
426};
