13def _lame_coefficients(dimension: int):
14 young_modulus, poisson_ratio = symbols("young_modulus poisson_ratio")
15 shear_modulus = young_modulus / (2 * (1 + poisson_ratio))
16 # dimension=2 is the plane-stress reduction of 3D isotropic elasticity;
17 # dimension=3 uses the three-dimensional Lame coefficient.
18 lame_lambda = (
19 young_modulus
20 * poisson_ratio
21 / ((1 + poisson_ratio) * (1 - (dimension - 1) * poisson_ratio))
22 )
23 return young_modulus, poisson_ratio, shear_modulus, lame_lambda

34class LinearElasticityHamiltonian:
35 @staticmethod
36 def __call__(dimension: int) -> HamiltonianDefinition:
37 if dimension < 1:
38 raise ValueError("dimension must be at least 1")
39
40 displacement = symbols(f"u0:{dimension}")
41 body_force = symbols(f"f0:{dimension}")
42 strain_indices = _strain_indices(dimension)
43 strain = symbols(f"strain0:{len(strain_indices)}")
44 _, _, shear_modulus, lame_lambda = _lame_coefficients(dimension)
45
46 strain_by_index = {
47 tensor_index: component
48 for tensor_index, component in zip(strain_indices, strain, strict=True)
49 }
50 trace_strain = sum(strain_by_index[i, i] for i in range(dimension))
51 strain_double_contraction = sum(
52 component**2 if i == j else 2 * component**2
53 for (i, j), component in strain_by_index.items()
54 )
55 strain_energy = (
56 shear_modulus * strain_double_contraction
57 + lame_lambda * trace_strain**2 / 2
58 )
59 body_force_potential = sum(
60 force_component * displacement_component
61 for force_component, displacement_component in zip(
62 body_force,
63 displacement,
64 strict=True,
65 )
66 )
67 hamiltonian = strain_energy - body_force_potential
68
69 return HamiltonianDefinition(
70 namespace="similie::physics::elasticity",
71 struct_name="LinearElasticityHamiltonian",
72 parameters=["young_modulus", "poisson_ratio"],
73 hamiltonian=hamiltonian,
74 variables=[displacement, strain, body_force],
75 includes=["<similie/physics/elasticity/elasticity_quantities.hpp>"],
76 template_parameters=["class... SpatialIndex"],
77 moments_object_component_expression="moments.template get<{index}>()",
78 component_indices=[
