26// Packed tensorial shear: xy = epsilon_xy, not engineering shear gamma_xy.
27// Hence dW/d(xy) = 2 sigma_xy; B^T dW/d(strain) already includes this
28// multiplicity. Only stress output divides the shear derivative by two.
29struct Strain2D
30{
31 double xx = 0.0;
32 double yy = 0.0;
33 double xy = 0.0;
34
35 template <class Index>
36 [[nodiscard]] constexpr double get() const
37 {
38 if constexpr (std::is_same_v<Index, StrainXX>) {
39 return xx;
40 } else if constexpr (std::is_same_v<Index, StrainXY>) {
41 return xy;
42 } else if constexpr (std::is_same_v<Index, StrainYY>) {
43 return yy;
44 } else {
45 static_assert(
46 std::is_same_v<Index, StrainXX> || std::is_same_v<Index, StrainYY>
47 || std::is_same_v<Index, StrainXY>,
48 "unsupported elasticity strain component index");
49 }
50 }
51};
