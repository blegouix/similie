118template <std::size_t N>
119class Topology
120{
121 static_assert(N > 0);
122 std::vector<std::array<Connection, 2 * N>> m_faces;
123 std::vector<BoundaryDomain> m_boundaries;
124
125 Connection& slot(Face face)
126 {
127 if (face.domain >= m_faces.size() || face.axis >= N)
128 throw std::invalid_argument("invalid physical face");
129 return m_faces[face.domain][2 * face.axis + (face.side == tensor::BoundarySide::Upper)];
130 }
131
132public:
133 explicit Topology(std::size_t domains) : m_faces(domains)
134 {
135 if (domains == 0)
136 throw std::invalid_argument("a multidomain needs a physical domain");
137 }
138 std::size_t domain_count() const
139 {
140 return m_faces.size();
141 }
142 std::size_t add_boundary(BoundaryDomain boundary)
143 {
144 m_boundaries.push_back(boundary);
145 return domain_count() + m_boundaries.size() - 1;
146 }
147 BoundaryDomain const& boundary(std::size_t id) const
148 {
149 return m_boundaries.at(id - domain_count());
150 }
151 void connect(
152 Face first,
153 Face second,
154 bool shared_boundary = true,
155 double orientation = 1.0,
156 double jump = 0.0)
157 {
158 Connection& a = slot(first);
159 Connection& b = slot(second);
160 if (first == second || first.axis != second.axis || !std::isfinite(orientation)
161 || orientation == 0.0 || !std::isfinite(jump) || !std::isfinite(1.0 / orientation)
162 || !std::isfinite(jump / orientation) || a.domain != Connection::UNCONNECTED
163 || b.domain != Connection::UNCONNECTED)
164 throw std::invalid_argument("invalid or duplicate interface connection");
165 a = {second.domain, second.side, shared_boundary, orientation, jump};
166 b = {first.domain, first.side, shared_boundary, 1.0 / orientation, -jump / orientation};
167 }
168 void connect_boundary(Face face, std::size_t id)
169 {
170 Connection& entry = slot(face);
171 if (id < domain_count() || id - domain_count() >= m_boundaries.size()
172 || entry.domain != Connection::UNCONNECTED)
173 throw std::invalid_argument("invalid or duplicate boundary connection");
174 entry.domain = id;
175 }
176 Connection const& connection(Face face) const
177 {
178 if (face.axis >= N)
179 throw std::invalid_argument("invalid face axis");
180 return m_faces.at(face.domain)[2 * face.axis + (face.side == tensor::BoundarySide::Upper)];
181 }
182 void validate() const
183 {
184 std::vector<bool> used(m_boundaries.size(), false);
185 for (std::array<Connection, 2 * N> const& faces : m_faces)
186 for (Connection const& edge : faces) {
187 if (edge.domain == Connection::UNCONNECTED)
188 throw std::invalid_argument("every physical face needs a connection");
189 if (edge.domain >= domain_count())
190 used.at(edge.domain - domain_count()) = true;
191 }
192 for (bool active : used)
193 if (!active)
194 throw std::invalid_argument("unconnected boundary node");
195 }
196};
