78// Unfold the canonical output components and structurally possible contractions.
79// Accessor conversions then see constant indices, including antisymmetric signs.
80// This works for a structured index in either operand, alongside other indices.
81template <class... Product, class... Left, class... Right, class... Contracted>
82class TensorProdStructured<
83 ddc::TypeSeq<Product...>,
84 ddc::TypeSeq<Left...>,
85 ddc::TypeSeq<Right...>,
86 ddc::TypeSeq<Contracted...>>
87{
88 template <std::size_t Output>
89 KOKKOS_FUNCTION static constexpr ddc::DiscreteElement<Product...> output_element()
90 {
91 std::array<std::size_t, sizeof...(Product)> const sizes {Product::mem_size()...};
92 std::array<std::size_t, sizeof...(Product)> ids {};
93 std::size_t remaining = Output;
94 for (std::size_t i = sizes.size(); i > 0; --i) {
95 ids[i - 1] = remaining % sizes[i - 1];
96 remaining /= sizes[i - 1];
97 }
98 ddc::DiscreteElement<Product...> element;
99 ddc::detail::array(element) = ids;
100 return element;
101 }

130 template <std::size_t Output, class Tensor1, class Tensor2, std::size_t... Values>
131 KOKKOS_FORCEINLINE_FUNCTION static void contract(
132 Tensor1 const& left,
133 Tensor2 const& right,
134 std::remove_const_t<typename Tensor1::element_type>& sum,
135 bool& initialized,
136 std::index_sequence<Values...> assigned)
137 {
138 constexpr auto known = tensor_product_known_element(
139 TensorAccessor<Product...>::canonical_natural_element(output_element<Output>()),
140 contracted_element(
141 std::index_sequence<Values...>(),
142 std::make_index_sequence<sizeof...(Values)>()));
143 if constexpr (
144 (tensor_product_indices_match<Left>(known) && ...)
145 && (tensor_product_indices_match<Right>(known) && ...)) {
146 if constexpr (sizeof...(Values) == sizeof...(Contracted)) {
147 constexpr auto left_element = TensorAccessor<Left...>::access_element(known);
148 constexpr auto right_element = TensorAccessor<Right...>::access_element(known);
149 if (initialized)
150 sum += left.get(left_element) * right.get(right_element);
151 else {
152 sum = left.get(left_element) * right.get(right_element);
153 initialized = true;
154 }
155 } else {
156 extend<Output>(
157 left,
158 right,
159 sum,
160 initialized,
161 assigned,
162 std::make_index_sequence<ddc::type_seq_element_t<
163 sizeof...(Values),
164 ddc::TypeSeq<Contracted...>>::size()>());
165 }
166 }
167 }

169 template <std::size_t Output, class ProductTensor, class Tensor1, class Tensor2>
170 KOKKOS_FORCEINLINE_FUNCTION static void write(
171 ProductTensor const& product,
172 Tensor1 const& left,
173 Tensor2 const& right)
174 {
175 constexpr ddc::DiscreteElement<Product...> element = output_element<Output>();
176 // Tensor views may select only a subset of the stored output components.
177 if (product.domain().contains(element)) {
178 typename ProductTensor::element_type sum = 0.;
179 bool initialized = false;
180 contract<Output>(left, right, sum, initialized, std::index_sequence<>());
181 product.mem(element) = sum;
182 }
183 }
