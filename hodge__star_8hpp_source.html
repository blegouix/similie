58template <std::size_t N, std::size_t M>
59KOKKOS_FUNCTION bool has_unique_ids(std::array<std::size_t, M> const& ids)
60{
61 if constexpr (M > 0) {
62 for (std::size_t i = 0; i < M; ++i) {
63 if (ids[i] >= N) {
64 return false;
65 }
66 for (std::size_t j = i + 1; j < M; ++j) {
67 if (ids[i] == ids[j]) {
68 return false;
69 }
70 }
71 }
72 }
73 return true;
74}

76template <std::size_t N, std::size_t M1, std::size_t M2>
77KOKKOS_FUNCTION int permutation_sign(
78 std::array<std::size_t, M1> const& source_ids,
79 std::array<std::size_t, M2> const& target_ids)
80{
81 std::array<std::size_t, N> permutation {};
82 for (std::size_t i = 0; i < M1; ++i) {
83 permutation[i] = source_ids[i];
84 }
85 if constexpr (M2 > 0) {
86 for (std::size_t i = 0; i < M2; ++i) {
87 permutation[M1 + i] = target_ids[i];
88 }
89 }
90
91 bool odd = false;
92 for (std::size_t i = 0; i < N; ++i) {
93 for (std::size_t j = i + 1; j < N; ++j) {
94 odd = (permutation[i] > permutation[j]) != odd;
95 }
96 }
97 return odd ? -1 : 1;
98}

100template <std::size_t N, std::size_t M>
101KOKKOS_FUNCTION std::array<std::size_t, N - M> complement_ids(std::array<std::size_t, M> const& ids)
102{
103 std::array<std::size_t, N - M> complement {};
104 std::size_t complement_id = 0;
105 for (std::size_t i = 0; i < N; ++i) {
106 bool found = false;
107 for (std::size_t id : ids) {
108 found = found || (id == i);
109 }
110 if (!found) {
111 complement[complement_id++] = i;
112 }
113 }
114 return complement;
115}

200 KOKKOS_FUNCTION static auto one_form_pairing(
201 MetricType metric,
202 PositionType position,
203 BatchElem elem)
204 {
205 using PositionIndex = ddc::
206 type_seq_element_t<0, ddc::to_type_seq_t<typename PositionType::indices_domain_t>>;
207 using MetricIndex = ddc::
208 type_seq_element_t<0, ddc::to_type_seq_t<typename MetricType::indices_domain_t>>;
209 constexpr std::size_t N = PositionIndex::size();
210 std::array<double, N * N> jacobian {}, gram {}, inverse {}, workspace {};
211 for (std::size_t axis = 0; axis < N; ++axis) {
212 std::array<double, N> const edge
213 = detail::edge_vector<PositionIndex>(position, elem, axis);
214 for (std::size_t physical = 0; physical < N; ++physical)
215 jacobian[physical * N + axis] = edge[physical];
216 }
217 for (std::size_t i = 0; i < N; ++i)
218 for (std::size_t j = 0; j < N; ++j)
219 for (std::size_t a = 0; a < N; ++a)
220 for (std::size_t b = 0; b < N; ++b)
221 gram[i * N + j]
222 += jacobian[a * N + i] * jacobian[b * N + j]
223 * metric.get(metric.access_element(
224 elem,
225 ddc::DiscreteElement<
226 tensor::metric_index_1<MetricIndex>,
227 tensor::metric_index_2<MetricIndex>>(a, b)));
228 auto gram_view = misc::math::
229 matrix_view<double, typename MetricType::memory_space>(gram.data(), N, N);
230 auto inverse_view = misc::math::
231 matrix_view<double, typename MetricType::memory_space>(inverse.data(), N, N);
232 if (!misc::math::
233 invert(inverse_view,
234 gram_view,
235 misc::math::vector_view<
236 double,
237 typename MetricType::memory_space>(workspace.data(), N * N)))
238 return inverse;
239 double const orientation = detail::reduction_determinant<N>(jacobian);
240 double const volume = (orientation < 0.0 ? -1.0 : 1.0)
241 * Kokkos::sqrt(Kokkos::abs(detail::reduction_determinant<N>(gram)));
242 for (double& coefficient : inverse)
243 coefficient *= volume;
244 return inverse;
245 }

298 KOKKOS_FUNCTION static double value(
299 MetricType metric,
300 PositionType position,
301 BatchElem elem,
302 auto natural_elem)
303 {
304 using SourceIndex = tensor::lower_t<reduction_index_t<Indices1>>;
305 using TargetIndex = reduction_index_t<Indices2>;
306 using SourceNaturalElem =
307 typename tensor::natural_domain_t<SourceIndex>::discrete_element_type;
308 using TargetNaturalElem =
309 typename tensor::natural_domain_t<TargetIndex>::discrete_element_type;
310 using SourceFullElem = misc::convert_type_seq_to_t<ddc::DiscreteElement, Indices1>;
311 using TargetFullElem = misc::convert_type_seq_to_t<ddc::DiscreteElement, Indices2>;
312
313 [[maybe_unused]] tensor::TensorAccessor<SourceIndex> source_accessor;
314 std::array<double, SourceIndex::access_size()> source_alloc {};
315 ddc::ChunkSpan<
316 double,
317 ddc::DiscreteDomain<SourceIndex>,
318 Kokkos::layout_right,
319 typename MetricType::memory_space>
320 source_span(source_alloc.data(), source_accessor.domain());
321 sil::tensor::Tensor source_tensor(source_span);
322 ddc::device_for_each(source_tensor.domain(), [&](auto source_mem_elem) {
323 source_tensor.mem(source_mem_elem) = 0.;
324 });
325
326 [[maybe_unused]] tensor::TensorAccessor<TargetIndex> target_accessor;
327 std::array<double, TargetIndex::access_size()> target_alloc {};
328 ddc::ChunkSpan<
329 double,
330 ddc::DiscreteDomain<TargetIndex>,
331 Kokkos::layout_right,
332 typename MetricType::memory_space>
333 target_span(target_alloc.data(), target_accessor.domain());
334 sil::tensor::Tensor target_tensor(target_span);
335 ddc::device_for_each(target_tensor.domain(), [&](auto target_mem_elem) {
336 target_tensor.mem(target_mem_elem) = 0.;
337 });
338
339 std::array source_ids = ddc::detail::array(SourceFullElem(natural_elem));
340 if constexpr (ddc::type_seq_size_v<Indices1> > 1) {
341 if (!detail::has_unique_ids<SourceIndex::size()>(source_ids)) {
342 return 0.;
343 }
344 }
345 bool odd = false;
346 for (auto i = source_ids.begin(); i < source_ids.end(); ++i) {
347 for (auto j = i + 1; j < source_ids.end(); ++j) {
348 odd = (*i > *j) != odd;
349 }
350 }
351 std::array canonical_source_ids(source_ids);
352 if constexpr (source_ids.size() > 1) {
353 for (auto i = canonical_source_ids.begin(); i < canonical_source_ids.end(); ++i) {
354 for (auto j = i + 1; j < canonical_source_ids.end(); ++j) {
355 if (*j < *i) {
356 Kokkos::kokkos_swap(*i, *j);
357 }
358 }
359 }
360 }
361
362 SourceNaturalElem source_natural_elem;
363 ddc::detail::array(source_natural_elem) = canonical_source_ids;
364 source_tensor(source_tensor.accessor().access_element(source_natural_elem)) = 1.;
365
366 operator()(target_tensor, source_tensor, metric, position, elem);
367
368 double const source_factor
369 = (odd ? -1. : 1.) / misc::factorial(ddc::type_seq_size_v<Indices1>);
370 if constexpr (ddc::type_seq_size_v<Indices2> == 0) {
371 return source_factor
372 * target_tensor(target_tensor.accessor().access_element(TargetNaturalElem()));
373 } else {
374 TargetNaturalElem target_natural_elem;
375 ddc::detail::array(target_natural_elem)
376 = ddc::detail::array(TargetFullElem(natural_elem));
377 return source_factor
378 * target_tensor.get(
379 target_tensor.accessor().access_element(target_natural_elem));
380 }
381 }
382};

384template <
385 misc::Specialization<ddc::TypeSeq> Indices1,
386 misc::Specialization<ddc::TypeSeq> Indices2,
387 class MetricType,
388 class BatchElem>
389struct ContinuousHodgeStar
390{
391 template <
392 misc::Specialization<tensor::Tensor> HodgeTensorType,
393 misc::Specialization<tensor::Tensor> FormTensorType>
394 KOKKOS_FUNCTION static void operator()(
395 HodgeTensorType hodge_tensor,
396 FormTensorType form_tensor,
397 MetricType metric,
398 BatchElem elem)
399 {
400 [[maybe_unused]] sil::tensor::tensor_accessor_for_domain_t<
401 hodge_star_domain_t<Indices1, Indices2>> hodge_star_accessor;
402 static constexpr std::size_t HODGE_STAR_SIZE
403 = misc::convert_type_seq_to_t<tensor::TensorFullIndex, Indices1>::access_size()
404 * detail::HodgeStarTargetSize<Indices2>::value;
405 std::array<double, HODGE_STAR_SIZE> hodge_star_alloc {};
406 ddc::ChunkSpan<
407 double,
408 hodge_star_domain_t<Indices1, Indices2>,
409 Kokkos::layout_right,
410 typename HodgeTensorType::memory_space>
411 hodge_star_span(hodge_star_alloc.data(), hodge_star_accessor.domain());
412 sil::tensor::Tensor hodge_star(hodge_star_span);
413 ddc::device_for_each(hodge_star.domain(), [&](auto it) {
414 hodge_star.mem(it) = value(metric, elem, hodge_star.canonical_natural_element(it));
415 });
416 sil::tensor::tensor_prod(hodge_tensor, hodge_star, form_tensor);
417 }
418
419 KOKKOS_FUNCTION static double value(MetricType metric, BatchElem elem, auto natural_elem)
420 {
421 constexpr std::size_t N = ddc::type_seq_size_v<Indices1> + ddc::type_seq_size_v<Indices2>;
422 constexpr std::size_t K = ddc::type_seq_size_v<Indices1>;
423 using SourceElem = misc::convert_type_seq_to_t<ddc::DiscreteElement, Indices1>;
424 using TargetElem = misc::convert_type_seq_to_t<ddc::DiscreteElement, Indices2>;
425 std::array const source_ids = ddc::detail::array(SourceElem(natural_elem));
426 std::array const target_ids = ddc::detail::array(TargetElem(natural_elem));
427
428 if (!detail::has_unique_ids<N>(source_ids) || !detail::has_unique_ids<N>(target_ids)) {
429 return 0.;
430 }
431
432 std::array<std::size_t, K> const complement = detail::complement_ids<N>(target_ids);
433 using MetricIndex = ddc::
434 type_seq_element_t<0, ddc::to_type_seq_t<typename MetricType::indices_domain_t>>;
435 using MetricIndex1 = tensor::metric_index_1<MetricIndex>;
436 using MetricIndex2 = tensor::metric_index_2<MetricIndex>;
437 std::array<double, N * N> metric_alloc {};
438 std::array<double, N * N> determinant_metric_alloc {};
439 std::array<double, N * N> inverse_metric_alloc {};
440 std::array<double, N * N> workspace_alloc {};
441 std::array<double, K * K> submatrix_alloc {};
442 for (std::size_t i = 0; i < N; ++i) {
443 for (std::size_t j = 0; j < N; ++j) {
444 metric_alloc[i * N + j] = metric.get(metric.access_element(
445 elem,
446 ddc::DiscreteElement<MetricIndex1, MetricIndex2>(i, j)));
447 }
448 }
449 determinant_metric_alloc = metric_alloc;
450
451 auto determinant_metric_view = misc::math::matrix_view<
452 double,
453 typename MetricType::memory_space>(determinant_metric_alloc.data(), N, N);
454 auto metric_view = misc::math::
455 matrix_view<double, typename MetricType::memory_space>(metric_alloc.data(), N, N);
456 double const determinant = misc::math::determinant(determinant_metric_view);
457 auto inverse_metric_view = misc::math::matrix_view<
458 double,
459 typename MetricType::memory_space>(inverse_metric_alloc.data(), N, N);
460 auto workspace = misc::math::vector_view<
461 double,
462 typename MetricType::memory_space>(workspace_alloc.data(), N * N);
463 if (!misc::math::invert(inverse_metric_view, metric_view, workspace)) {
464 return 0.;
465 }
466
467 return Kokkos::sqrt(Kokkos::abs(determinant))
468 * static_cast<double>(detail::permutation_sign<N>(complement, target_ids))
469 * misc::math::submatrix_determinant<
470 K>(inverse_metric_view, source_ids, complement, submatrix_alloc)
471 / misc::factorial(K);
472 }
473};

595template <std::size_t Axis, class Position, class Rules, class Element>
596KOKKOS_FUNCTION void geometry_front(
597 Position const& position,
598 Rules const& rules,
599 Element elem,
600 Element& front)
601{
602 if (!tensor::detail::is_interface(rules.template boundary_rule<Axis, true>())) {
603 if (ddc::detail::array(elem)[Axis]
604 == ddc::detail::array(position.non_indices_domain().back())[Axis]
605 && ddc::detail::array(elem)[Axis]
606 > ddc::detail::array(position.non_indices_domain().front())[Axis])
607 --ddc::detail::array(front)[Axis];
608 }
609}

611template <class Position, class Rules, class Element, std::size_t... Axis>
612KOKKOS_FUNCTION auto local_geometry(
613 Position const& position,
614 Rules const& rules,
615 Element elem,
616 std::index_sequence<Axis...>)
617{
618 using Index
619 = ddc::type_seq_element_t<0, ddc::to_type_seq_t<typename Position::indices_domain_t>>;
620 Element front = elem;
621 (geometry_front<Axis>(position, rules, elem, front), ...);
622 auto geometry = make_stencil<typename Position::memory_space, Index>(front);
623 ddc::device_for_each(geometry.domain(), [&](auto sample) {
624 geometry.mem(sample)
625 = rules(position, Element(sample)).mem(ddc::DiscreteElement<Index>(sample));
626 });
627 return geometry;
628}

730 template <class Element>
731 KOKKOS_FUNCTION auto cell_geometry(Element cell) const
732 {
733 constexpr std::size_t N = Direction::size();

735 using Index = ddc::
736 type_seq_element_t<0, ddc::to_type_seq_t<typename Position::indices_domain_t>>;
737 auto positions = detail::make_stencil<typename Position::memory_space, Index>(cell);
738 std::array<double, N * N> jacobian {};
739 for (std::size_t axis = 0; axis < N; ++axis) {
740 for (std::size_t edge_id = 0; edge_id < edge_count; ++edge_id) {
741 Element first = cell;
742 std::size_t tangent = 0;
743 for (std::size_t other = 0; other < N; ++other)
744 if (other != axis)
745 ddc::detail::array(first)[other] += (edge_id >> tangent++) & 1;
746 Element second = first;
747 ++ddc::detail::array(second)[axis];
748 for (std::size_t physical = 0; physical < N; ++physical)
749 jacobian[physical * N + axis]
750 += (geometry(position, second)
751 .mem(ddc::DiscreteElement<Index>(physical))
752 - geometry(position, first)
753 .mem(ddc::DiscreteElement<Index>(physical)))
754 / edge_count;
755 }
756 }
757 ddc::device_for_each(positions.domain(), [&](auto sample) {
758 std::size_t const physical = ddc::DiscreteElement<Index>(sample).uid();
759 double value = 0.0;
760 for (std::size_t axis = 0; axis < N; ++axis)
761 value += jacobian[physical * N + axis]
762 * (ddc::detail::array(Element(sample))[axis]
763 - ddc::detail::array(cell)[axis]);
764 positions.mem(sample) = value;
765 });
766 return positions;
767 }

842 template <class Element, class Component, std::size_t... Axis>
843 KOKKOS_FUNCTION auto stencil(Element elem, Component component, std::index_sequence<Axis...>)
844 const
845 {
846 using Index = ddc::
847 type_seq_element_t<0, ddc::to_type_seq_t<typename Cochain::indices_domain_t>>;
848 constexpr std::size_t N = Direction::size();
849 constexpr std::size_t edge_count = std::size_t(1) << (N - 1);
850 auto result = detail::make_stencil<typename Position::memory_space, Index, 3>(Element {});
851 double prescribed = 0.0;
852 if (prescribed_flux(elem, component, prescribed))
853 return result;
854 std::size_t const normal = normal_direction(component);
855 tensor::TensorAccessor<component_type> accessor;
856 auto const ids
857 = detail::flat_natural_elem_ids(accessor.canonical_natural_element(component));
858 auto metric = detail::euclidean_metric<typename Position::memory_space>(
859 typename tensor::uncharacterize_t<Direction>::type_seq_dimensions {});
860 for (std::size_t incident = 0; incident < edge_count; ++incident) {
861 Element cell = elem;
862 std::size_t bit = 0;
863 for (std::size_t axis = 0; axis < N; ++axis)
864 if (axis != normal)
865 ddc::detail::array(cell)[axis] -= (incident >> bit++) & 1;
866 if (!(physical_cell<Axis>(cell) && ...))
867 continue;
868 auto positions = cell_geometry(cell);
869 tensor::Tensor<
870 double,
871 typename Position::discrete_domain_type,
872 Kokkos::layout_right,
873 typename Position::memory_space>
874 local_position(positions);
875 std::array<double, N * N> const pairing = DiscreteHodgeStar<
876 CellComplex::Primal,
877 source_indices,
878 dual_indices,
879 decltype(metric),
880 decltype(local_position),
881 Element>::one_form_pairing(metric, local_position, cell);
882 using PositionIndex = ddc::
883 type_seq_element_t<0, ddc::to_type_seq_t<typename Position::indices_domain_t>>;
884 std::array<double, N * edge_count * N> edges {};
885 std::array<double, N * edge_count> projector_row {};
886 std::array<double, N> weighted_projection {};
887 std::array<double, N * N> gram {}, inverse {}, workspace {};
888 std::size_t selected = 0;
889 for (std::size_t axis = 0; axis < N; ++axis) {
890 for (std::size_t edge_id = 0; edge_id < edge_count; ++edge_id) {
891 Element first = cell;
892 std::size_t tangent = 0;
893 for (std::size_t other = 0; other < N; ++other)
894 if (other != axis)
895 ddc::detail::array(first)[other] += (edge_id >> tangent++) & 1;
896 Element second = first;
897 ++ddc::detail::array(second)[axis];
898 std::size_t const edge = axis * edge_count + edge_id;
899 if (axis == normal && first == elem)
900 selected = edge;
901 for (std::size_t physical = 0; physical < N; ++physical)
902 edges[edge * N + physical]
903 = geometry(position, second)
904 .mem(ddc::DiscreteElement<PositionIndex>(physical))
905 - geometry(position, first)
906 .mem(ddc::DiscreteElement<PositionIndex>(physical));
907 for (std::size_t i = 0; i < N; ++i) {
908 for (std::size_t j = 0; j < N; ++j)
909 gram[i * N + j] += edges[edge * N + i] * edges[edge * N + j];
910 }
911 }
912 }
913 auto inverse_view = misc::math::
914 matrix_view<double, typename Position::memory_space>(inverse.data(), N, N);
915 if (!misc::math::
916 invert(inverse_view,
917 misc::math::matrix_view<
918 double,
919 typename Position::memory_space>(gram.data(), N, N),
920 misc::math::vector_view<
921 double,
922 typename Position::memory_space>(workspace.data(), N * N)))
923 continue;
924
925 // One row of P W P, with P = I - E (E^T E)^-1 E^T.
926 // It annihilates constant Cartesian forms, as in the pointwise Hodge.
927 for (std::size_t edge = 0; edge < N * edge_count; ++edge) {
928 projector_row[edge] = edge == selected ? 1.0 : 0.0;
929 for (std::size_t i = 0; i < N; ++i)
930 for (std::size_t j = 0; j < N; ++j)
931 projector_row[edge] -= edges[selected * N + i] * inverse[i * N + j]
932 * edges[edge * N + j];
933 projector_row[edge] *= pairing[(edge / edge_count) * (N + 1)] / edge_count;
934 for (std::size_t i = 0; i < N; ++i)
935 weighted_projection[i] += edges[edge * N + i] * projector_row[edge];
936 }
937 for (std::size_t axis = 0; axis < N; ++axis)
938 for (std::size_t edge_id = 0; edge_id < edge_count; ++edge_id) {
939 std::size_t const edge = axis * edge_count + edge_id;
940 double weight = pairing[normal * N + axis] / (edge_count * edge_count)
941 + projector_row[edge];
942 for (std::size_t i = 0; i < N; ++i)
943 for (std::size_t j = 0; j < N; ++j)
944 weight -= weighted_projection[i] * inverse[i * N + j]
945 * edges[edge * N + j];
946 weight *= diffusivity
947 * detail::permutation_sign<
948 N>(std::array<std::size_t, 1> {normal}, ids);
949 Element first = cell;
950 std::size_t tangent = 0;
951 for (std::size_t other = 0; other < N; ++other)
952 if (other != axis)
953 ddc::detail::array(first)[other] += (edge_id >> tangent++) & 1;
954 Element local = first;
955 for (std::size_t coordinate = 0; coordinate < N; ++coordinate)
956 ddc::detail::array(local)[coordinate]
957 = ddc::detail::array(first)[coordinate]
958 - ddc::detail::array(elem)[coordinate] + 1;
959 if constexpr (Index::rank() == 0) {
960 // The scalar coboundary's tensor stencil is the oriented
961 // endpoint difference; retain that incidence independently of H.
962 result.mem(local, ddc::DiscreteElement<Index>(0)) -= weight;
963 ++ddc::detail::array(local)[axis];
964 result.mem(local, ddc::DiscreteElement<Index>(0)) += weight;
965 } else
966 result.mem(local, ddc::DiscreteElement<Index>(axis)) += weight;
967 }
968 }
969 return result;
970 }
