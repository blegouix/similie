148struct NaturalScalarExtrapolationRule
149{
150 static constexpr bool STORED_SAMPLES_ONLY = true;
151 template <class Domain, class Element>
152 KOKKOS_FUNCTION auto value(Domain domain, Element element) const
153 {
154 auto result = detail::make_extrapolation_basis(domain, element);
155 misc::convert_type_seq_to_t<misc::SignedDiscreteElement, ddc::to_type_seq_t<Domain>> const
156 sample(element),
157 nearest(misc::clamp_to_domain(domain, sample)), front(domain.front()),
158 back(domain.back());
159 result.mem(nearest.storage_element(), ddc::DiscreteElement<detail::ExtrapolationSource>(0))
160 = 1.0;
161 for (std::size_t dim = 0; dim < ddc::type_seq_size_v<ddc::to_type_seq_t<Domain>>; ++dim) {
162 std::ptrdiff_t const offset
163 = static_cast<std::ptrdiff_t>(ddc::detail::array(sample)[dim])
164 - static_cast<std::ptrdiff_t>(ddc::detail::array(nearest)[dim]);
165 if (offset == 0 || ddc::detail::array(front)[dim] == ddc::detail::array(back)[dim])
166 continue;
167 misc::convert_type_seq_to_t<misc::SignedDiscreteElement, ddc::to_type_seq_t<Domain>>
168 inner(nearest);
169 ddc::detail::array(inner)[dim] += offset < 0 ? 1 : -1;
170 double const distance = offset < 0 ? -double(offset) : double(offset);
171 result
172 .mem(nearest.storage_element(),
173 ddc::DiscreteElement<detail::ExtrapolationSource>(0))
174 += distance;
175 result
176 .mem(inner.storage_element(),
177 ddc::DiscreteElement<detail::ExtrapolationSource>(0))
178 -= distance;
179 }
180 return result;
181 }
182 template <class Field, class Element>
183 KOKKOS_FUNCTION auto operator()(Field const& field, Element element) const
184 {
185 auto const basis = value(field.non_indices_domain(), element);
186 return detail::pointwise_tensor<Field>([&](auto component) {
187 return detail::contract_extrapolation_basis(
188 basis,
189 [&](auto sample) { return field.mem(sample, component); },
190 [](std::size_t) { return 0.0; });
191 });
192 }
193};
