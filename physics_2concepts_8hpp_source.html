41// A complete state keeps coupled components available to every operation.
42struct ConceptHamiltonian
43{
44 static constexpr std::size_t N = 1;
45 static constexpr bool IS_LINEAR = true;
46 struct State
47 {
48 std::array<double, 1> potential {};
49 std::array<double, N> moments {};
50 };
51
52 template <class Element>
53 KOKKOS_FUNCTION constexpr double hamiltonian(State const& state, Element) const
54 {
55 return (state.potential[0] * state.potential[0] + state.moments[0] * state.moments[0]) / 2;
56 }
57 template <class Index, class Element>
58 KOKKOS_FUNCTION constexpr double dhamiltonian_dmoments(State const& state, Element) const
59 {
60 return state.moments[Index::value];
61 }
62 template <class Index, class Element>
63 KOKKOS_FUNCTION constexpr double dhamiltonian_dpotential(State const& state, Element) const
64 {
65 return state.potential[Index::value];
66 }
67 template <class Row, class Column, class Element>
68 KOKKOS_FUNCTION constexpr double jacobian(State const&, Element) const
69 {
70 return Row::value == Column::value ? 1 : 0;
71 }
72 template <class Index, class Element>
73 KOKKOS_FUNCTION constexpr double dhamiltonian_dmoments_value(State const&, Element) const
74 {
75 return 1;
76 }
77 template <class Index, class Element>
78 KOKKOS_FUNCTION constexpr double dhamiltonian_dpotential_value(State const&, Element) const
79 {
80 return 1;
81 }
82};

100// Every Hamiltonian has the same arguments; quantities belong to its State.
101template <class Hamiltonian, class Element>
102concept HamiltonianConcept = requires(
103 Hamiltonian const& h,
104 typename Hamiltonian::State const& state,
105 Element element) {
106 requires std::same_as<std::remove_cv_t<decltype(Hamiltonian::N)>, std::size_t>;
107 requires std::same_as<std::remove_cv_t<decltype(Hamiltonian::IS_LINEAR)>, bool>;
108 typename std::integral_constant<std::size_t, Hamiltonian::N>;
109 typename std::bool_constant<Hamiltonian::IS_LINEAR>;
110 { h.hamiltonian(state, element) } -> std::same_as<double>;
111 {
112 h.template dhamiltonian_dmoments<std::integral_constant<std::size_t, 0>>(state, element)
113 } -> std::same_as<double>;
114 {
115 h.template dhamiltonian_dpotential<std::integral_constant<std::size_t, 0>>(state, element)
116 } -> std::same_as<double>;
117 {
118 h.template jacobian<
119 std::integral_constant<std::size_t, 0>,
120 std::integral_constant<std::size_t, 0>>(state, element)
121 } -> std::same_as<double>;
122};

188// Scalar and vector laws use arrays of the same form, with N components.
189template <class Law, class Element>
190concept ConstitutiveLawConcept
191 = requires(Law const& law, typename Law::State const& state, Element element) {
192 typename std::integral_constant<std::size_t, Law::N>;
193 { law(state, element) } -> std::same_as<std::array<double, Law::N>>;
194 { law.value(state, element) } -> std::same_as<std::array<double, Law::N * Law::N>>;
195 { law.jacobian(state, element) } -> std::same_as<std::array<double, Law::N * Law::N>>;
196 { law.inverse(state, element) } -> std::same_as<std::array<double, Law::N>>;
197 {
198 law.inverse_value(state, element)
199 } -> std::same_as<std::array<double, Law::N * Law::N>>;
200 };
