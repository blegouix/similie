16template <tensor::TensorIndex Index, class SimplexType>
17KOKKOS_FUNCTION auto integration_element(SimplexType const& simplex)
18{
19 static_assert(Index::rank() == SimplexType::dimension());
20 std::array<std::size_t, Index::rank()> directions {};
21 std::size_t count = 0;
22 auto const vector = ddc::detail::array(simplex.discrete_vector());
23 for (std::size_t axis = 0; axis < vector.size(); ++axis) {
24 if (vector[axis] != 0) {
25 assert(count < directions.size());
26 directions[count++] = axis;
27 }
28 }
29 assert(count == directions.size());
30 if constexpr (Index::rank() == 0) {
31 return typename tensor::natural_domain_t<Index>::discrete_element_type(0);
32 } else {
33 return natural_elem_from_flat_ids<
34 typename tensor::natural_domain_t<Index>::discrete_element_type>(directions);
35 }
36}

48template <tensor::TensorIndex Index, class CochainTensor, class SimplexType>
49KOKKOS_FUNCTION double integrate(CochainTensor const& cochain, SimplexType const& simplex)
50{
51 return (simplex.negative() ? -1.0 : 1.0)
52 * cochain.get(cochain.access_element(
53 simplex.discrete_element(),
54 detail::integration_element<Index>(simplex)));
55}

89template <misc::Specialization<ddc::TypeSeq> Indices, class Form, class Position, class SimplexType>
90KOKKOS_FUNCTION double integrate(
91 Form const& form,
92 Position const& position,
93 SimplexType const& simplex)
94{
95 assert(misc::domain_contains(form.non_indices_domain(), simplex.discrete_element()));
96 assert(misc::domain_contains(position.non_indices_domain(), simplex.discrete_element()));
97 assert(misc::domain_contains(
98 position.non_indices_domain(),
99 simplex.discrete_element() + simplex.discrete_vector()));
100 using Operator = Reduction<Indices, Position, typename SimplexType::discrete_element_type>;
101 using NaturalElement = typename tensor::tensor_accessor_for_domain_t<
102 reduction_domain_t<Indices>>::natural_domain_t::discrete_element_type;
103 if constexpr (ddc::type_seq_size_v<Indices> == 0) {
104 return integrate<typename Operator::target_index_type>(form, simplex);
105 } else {
106 // Reduce only this cell's tangent component. Other forward directions may
107 // lie outside the domain, for example on an upper boundary face.
108 auto const target = detail::flat_natural_elem_ids(
109 detail::integration_element<typename Operator::target_index_type>(simplex));
110 [[maybe_unused]] tensor::TensorAccessor<typename Operator::source_index_type> source;
111 double result = 0.0;
112 ddc::device_for_each(source.natural_domain(), [&](auto source_element) {
113 auto const coefficient = detail::merge_reduction_natural_elems<
114 NaturalElement>(detail::flat_natural_elem_ids(source_element), target);
115 using FormElement = typename Form::accessor_t::natural_domain_t::discrete_element_type;
116 FormElement const form_element = detail::natural_elem_from_flat_ids<FormElement>(
117 detail::flat_natural_elem_ids(source_element));
118 result += Operator::value(position, simplex.discrete_element(), coefficient)
119 * form.get(form.access_element(simplex.discrete_element(), form_element));
120 });
121 return (simplex.negative() ? -1.0 : 1.0) * result;
122 }
123}
