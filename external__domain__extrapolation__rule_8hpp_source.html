37template <class Dimension>
38struct ExternalDomainBoundaryMap
39{
40 BoundarySide source_side;
41 BoundarySide target_side;
42 bool shared_boundary = true;
43
44private:
45 template <class DDim, class SourceDomain, class TargetDomain, class Element>
46 KOKKOS_FUNCTION void map_coordinate(
47 SourceDomain source,
48 TargetDomain const& target,
49 Element sampled,
50 Element& mapped) const
51 {
52 if constexpr (std::is_same_v<DDim, Dimension>) {
53 ddc::DiscreteElement<Dimension> const source_boundary(
54 source_side == BoundarySide::Lower ? source.front() : source.back());
55 ddc::DiscreteElement<Dimension> const target_boundary(
56 target_side == BoundarySide::Lower ? target.front() : target.back());
57 std::ptrdiff_t const offset
58 = static_cast<std::ptrdiff_t>(sampled.template uid<Dimension>())
59 - static_cast<std::ptrdiff_t>(source_boundary.template uid<Dimension>());
60 // Signed differences also handle unsigned elements wrapping below zero.
61 std::ptrdiff_t const distance = source_side == BoundarySide::Lower ? -offset : offset;
62 assert(distance >= 0);
63 std::ptrdiff_t const inward = distance - (shared_boundary ? 0 : 1);
64 assert(inward >= 0);
65 mapped.template uid<Dimension>()
66 = static_cast<std::ptrdiff_t>(target_boundary.template uid<Dimension>())
67 + (target_side == BoundarySide::Lower ? inward : -inward);
68 } else {
69 assert(source.template extent<DDim>() == target.template extent<DDim>());
70 std::ptrdiff_t const offset
71 = static_cast<std::ptrdiff_t>(sampled.template uid<DDim>())
72 - static_cast<std::ptrdiff_t>(source.front().template uid<DDim>());
73 mapped.template uid<DDim>()
74 = static_cast<std::ptrdiff_t>(target.front().template uid<DDim>()) + offset;
75 }
76 }
77
78public:
79 template <class... DDim, class TargetDomain, class Element>
80 KOKKOS_FUNCTION Element operator()(
81 ddc::DiscreteDomain<DDim...> const& source,
82 TargetDomain const& target,
83 Element sampled) const
84 {
85 static_assert((std::is_same_v<Dimension, DDim> || ...));
86 static_assert(std::is_same_v<ddc::TypeSeq<DDim...>, ddc::to_type_seq_t<TargetDomain>>);
87 Element mapped = sampled;
88 (map_coordinate<DDim>(source, target, sampled, mapped), ...);
89 return mapped;
90 }

92 template <class SourceDomain, class TargetDomain, class Element>
93 KOKKOS_FUNCTION Element
94 map_boundary(SourceDomain source, TargetDomain const& target, Element sampled) const
95 {
96 assert(shared_boundary);
97 assert(sampled.template uid<Dimension>()
98 == (source_side == BoundarySide::Lower ? source.front() : source.back())
99 .template uid<Dimension>());
100 return (*this)(source, target, sampled);
101 }

132 {

134 result
135 .mem(detail::extrapolation_sample(domain, element),
136 ddc::DiscreteElement<detail::ExtrapolationSource>(
137 misc::domain_contains(domain, element) ? 0 : 1))
138 = 1.0;

140 }

142 template <class Field, class Element>
143 KOKKOS_FUNCTION auto operator()(Field const& field, Element element) const
144 {
145 if (misc::domain_contains(field.non_indices_domain(), element))
146 return detail::pointwise_tensor<Field>([&](auto component) {
147 return field.mem(detail::storage_element(element), component);
148 });
149 auto const donor = target_rule(
150 neighbor,
151 boundary_map(field.non_indices_domain(), neighbor.non_indices_domain(), element));
152 return detail::pointwise_tensor<Field>(
153 [&](auto component) { return orientation * donor.mem(component) + jump; });
154 }
