181template <
182 CellComplex Complex,
183 class Indices1,

188class WeightedLeviCivitaDiscreteHodgeStar
189{
190 using Source = tensor::lower_t<reduction_index_t<Indices1>>;
191 using Target = reduction_index_t<Indices2>;
192 using Index = detail::WeightedHodgeIndex<Indices1, Indices2>;
193 using MetricIndex
194 = ddc::type_seq_element_t<0, ddc::to_type_seq_t<typename MetricType::indices_domain_t>>;
195 static_assert(misc::Specialization<MetricIndex, tensor::TensorIdentityIndex>);
196 static_assert(tensor::metric_index_1<MetricIndex>::size() == Index::rank());
197
198 KOKKOS_FUNCTION static std::array<double, Index::mem_size()> weights(
199 MetricType const& metric,
200 PositionType const& position,
201 BatchElem element)
202 {
203 auto const reconstruction = Reconstruction<
204 tensor::lower_t<Indices1>,
205 PositionType,
206 BatchElem,
207 CellComplex::Primal,
208 tensor::TensorDiagonalIndex>::value(position, element);
209 auto const reduction = Reduction<
210 Indices2,
211 PositionType,
212 BatchElem,
213 Complex,
214 tensor::TensorDiagonalIndex>::value(metric, position, element);
215 std::array<double, Index::mem_size()> result {};
216 for (std::size_t entry = 0; entry < result.size(); ++entry) {
217 std::array const components
218 = detail::weighted_hodge_components<Indices1, Indices2>(entry);
219 result[entry]
220 = reconstruction.weights[components[0] * Source::mem_size() + components[0]]
221 * reduction.weights[components[1] * Target::mem_size() + components[1]]
222 * misc::permutation_parity(Index::mem_id_to_canonical_natural_ids(entry));
223 }
224 return result;
225 }
