6"""Derive local electromagnetic exchange from the minimal-coupling Hamiltonian.
7
8H(x,p,t) = |p-q A(x,t)|^2/(2m) + q phi(x,t).
9Hamilton's equations evolve canonical momentum p. The force on mechanical
10momentum m v = p-q A additionally subtracts q dA/dt along the trajectory.
11After introducing E=-grad(phi)-partial_t A and B=curl(A), this gives the
12electric and magnetic forces. Summing carriers replaces q by charge density
13and q v by total current density; no common carrier velocity is assumed.
14
15These are exchange laws, not extra stored energies to add to fluid+field H.
16Magnetic work vanishes for each carrier. All fluid energy transfer is J.E,
17including conversion into internal energy for a conducting fluid.
18"""

23from sympy import Function, Matrix, Symbol, diff, simplify, symbols
24from sympy.printing.codeprinter import cxxcode

27class ElectromagneticInteraction:
28 def __init__(self):
29 self.position = Matrix(symbols("x0:3", real=True))
30 self.time = symbols("t", real=True)
31 self.momentum = Matrix(symbols("p0:3", real=True))
32 self.velocity = Matrix(symbols("v0:3", real=True))
33 self.charge, self.mass = symbols("q m", nonzero=True, real=True)
34 self.potential = Function("phi")(*self.position, self.time)
35 self.vector_potential = Matrix(
36 [Function(f"A{i}")(*self.position, self.time) for i in range(3)]
37 )
38 mechanical = self.momentum - self.charge * self.vector_potential
39 self.hamiltonian = mechanical.dot(mechanical) / (2 * self.mass)
40 self.hamiltonian += self.charge * self.potential
41 self.canonical_velocity = self.hamiltonian.diff(self.momentum)
42 self.canonical_force = -self.hamiltonian.diff(self.position)
43 mechanical_force = self.canonical_force - self.charge * (
44 self.vector_potential.diff(self.time)
45 + self.vector_potential.jacobian(self.position) * self.canonical_velocity
46 )
47 mechanical_force = mechanical_force.subs(
48 dict(
49 zip(
50 self.momentum,
51 self.mass * self.velocity + self.charge * self.vector_potential,
52 )
53 ),
54 simultaneous=True,
55 )
56
57 self.electric_field = Matrix(symbols("E0:3", real=True))
58 self.induction = Matrix(symbols("B0:3", real=True))
59 substitutions = {
60 diff(self.potential, self.position[i]): -self.electric_field[i]
61 - diff(self.vector_potential[i], self.time)
62 for i in range(3)
63 }
64 # Rewrite the three antisymmetric derivative pairs using B=curl A.
65 # The remaining symmetric derivatives cancel, without choosing a gauge.
66 for i in range(3):
67 j, k = (i + 1) % 3, (i + 2) % 3
68 substitutions[diff(self.vector_potential[k], self.position[j])] = (
69 self.induction[i] + diff(self.vector_potential[j], self.position[k])
70 )
71 self.mechanical_force = mechanical_force.subs(
72 substitutions, simultaneous=True
73 ).applyfunc(simplify)
74 self.electric_force = self.mechanical_force.subs(
75 dict.fromkeys(self.velocity, 0)
76 )
77 self.magnetic_force = self.mechanical_force - self.electric_force
78 self.electric_power = self.velocity.dot(self.electric_force)
79 self.magnetic_power = simplify(self.velocity.dot(self.magnetic_force))
80 self.charge_density = symbols("charge_density", real=True)
81 self.current = Matrix(symbols("J0:3", real=True))
82 carrier_sum = {
83 self.velocity[i]: self.current[i] / self.charge for i in range(3)
84 }
85 self.electric_force_density = self.electric_force.subs(
86 self.charge, self.charge_density
87 )
88 self.magnetic_force_density = self.magnetic_force.subs(carrier_sum).applyfunc(
89 simplify
90 )
91 self.fluid_energy_source = simplify(self.electric_power.subs(carrier_sum))

94def generate(output_directory):
95 model = ElectromagneticInteraction()
