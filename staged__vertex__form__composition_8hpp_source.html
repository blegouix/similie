13// Each source k-face contributes through its 2^k vertices. Retain those
14// coefficients rather than expanding the product of the two spatial supports.
15template <class Index, class Domain, class Memory>
16struct VertexTransferFactors
17{
18 static constexpr std::size_t corners = std::size_t(1) << Index::rank();
19 using Element = typename Domain::discrete_element_type;
20 using Local = tensor::Tensor<double, ddc::DiscreteDomain<Index>, Kokkos::layout_right, Memory>;
21 Kokkos::View<
22 double * [Index::mem_size() * Index::mem_size() * corners],
23 Kokkos::LayoutLeft,
24 Memory>
25 weights;

27 KOKKOS_FUNCTION static Element sample(
28 Element node,
29 std::size_t component,
30 std::size_t corner,
31 int sign)
32 {
33 tensor::TensorAccessor<Index> accessor;
34 auto const natural
35 = accessor.canonical_natural_element(ddc::DiscreteElement<Index>(component));
36 auto const axes = flat_natural_elem_ids(natural);
37 for (std::size_t axis = 0; axis < axes.size(); ++axis)
38 ddc::detail::array(node)[axes[axis]] += sign * int((corner >> axis) & 1);
39 return node;
40 }

42 struct UnitSample
43 {
44 Element node;
45 std::size_t component;
46 template <class Field, class Point>
47 KOKKOS_FUNCTION auto operator()(Field const&, Point point) const
48 {
49 return tensor::detail::pointwise_tensor<ddc::DiscreteDomain<Index>, Memory>(
50 [&](auto c) {
51 return point == node && c.template uid<Index>() == component ? 1.0 : 0.0;
52 });
53 }
54 };

56 template <bool Projection, class Exec, class Field, class Position>
57 static VertexTransferFactors build(
58 Exec const& exec,
59 Field const& field,
60 Position const& position)
61 {
62 VertexTransferFactors result;
63 Domain const domain = field.non_indices_domain();
64 result.weights = decltype(weights)(
65 Kokkos::view_alloc(Kokkos::WithoutInitializing, "staged_vertex_transfer_factors"),
66 domain.size());
67 auto const weights = result.weights;
68 using Indices = ddc::to_type_seq_t<tensor::natural_domain_t<Index>>;
69 Kokkos::parallel_for(
70 "similie_cache_vertex_transfer_factors",
71 Kokkos::RangePolicy<Exec>(exec, 0, domain.size()),
72 KOKKOS_LAMBDA(std::size_t node) {
73 (void)weights;
74 (void)field;
75 (void)position;
76 Element const element = stencil_element(domain, node);
77 std::array<double, Index::mem_size()> buffer {};
78 Local local(
79 ddc::ChunkSpan<
80 double,
81 ddc::DiscreteDomain<Index>,
82 Kokkos::layout_right,
83 Memory>(
84 buffer.data(),
85 tensor::TensorAccessor<Index>::domain()));
86 for (std::size_t source = 0; source < Index::mem_size(); ++source)
87 for (std::size_t corner = 0; corner < corners; ++corner) {
88 // Projection support follows the output face; reconstruction
89 // support follows the source face. Probe the canonical operators.
90 if constexpr (!Projection) {
91 VertexFormReconstruction<Indices>::operator()(
92 Form(local),
93 field,
94 position,
95 element,
96 UnitSample {sample(element, source, corner, -1), source});
97 for (std::size_t target = 0; target < Index::mem_size(); ++target)
98 weights(node,
99 (target * Index::mem_size() + source) * corners
100 + corner)
101 = local.mem(ddc::DiscreteElement<Index>(target));
102 } else {
103 for (std::size_t target = 0; target < Index::mem_size(); ++target) {
104 VertexFormProjection<Indices>::operator()(
105 Cochain(local),
106 field,
107 position,
108 element,
109 UnitSample {
110 sample(element, target, corner, 1),
111 source});
112 weights(node,
113 (target * Index::mem_size() + source) * corners
114 + corner)
115 = local.mem(ddc::DiscreteElement<Index>(target));
116 }
117 }
118 }
119 });
120 return result;
121 }

123 template <class Output, class Field>
124 KOKKOS_FUNCTION void reconstruct(
125 Output const& output,
126 Field const& field,
127 Domain const& domain,
128 Element element) const
129 {
130 std::size_t const node = stencil_node(domain, element);
131 for (std::size_t target = 0; target < Index::mem_size(); ++target) {
132 double value = 0;
133 for (std::size_t source = 0; source < Index::mem_size(); ++source)
134 for (std::size_t corner = 0; corner < corners; ++corner) {
135 auto const point = tensor::PeriodicExtrapolationRule {}(
136 field,
137 sample(element, source, corner, -1));
138 value += weights(node, (target * Index::mem_size() + source) * corners + corner)
139 * point.mem(ddc::DiscreteElement<Index>(source));
140 }
141 output.mem(ddc::DiscreteElement<Index>(target)) = value;
142 }
143 }

227 ddc::ChunkSpan<

231 Memory>(

234 for (std::size_t target = 0; target < Target::mem_size(); ++target) {

236 for (std::size_t corner = 0;
237 corner < decltype(storage.projection)::corners;
238 ++corner) {
239 auto const sample = decltype(storage.projection)::
240 sample(element, target, corner, 1);
241 // Apply the same periodic observation rule as the field adapter.
242 auto wrapped = sample;
243 for (std::size_t axis = 0; axis < Domain::rank(); ++axis) {
244 std::ptrdiff_t const front
245 = ddc::detail::array(storage.domain.front())[axis];
246 std::ptrdiff_t const count
247 = ddc::detail::array(storage.domain.extents())[axis];
248 std::ptrdiff_t const relative
249 = ddc::detail::array(sample)[axis] - front;
250 ddc::detail::array(wrapped)[axis]
251 = front + (relative % count + count) % count;
252 }
253 storage.first_factors
254 .reconstruct(first, a, storage.domain, wrapped);
255 storage.second_factors
256 .reconstruct(second, b, storage.domain, wrapped);
257 Operator {}(
258 Form(transformed),
259 Form(first),
260 Form(second),
261 storage.inverse_metric,
262 wrapped);
263 for (std::size_t source = 0; source < Target::mem_size(); ++source)
264 value += storage.projection.weights(
265 node,
266 (target * Target::mem_size() + source)
267 * decltype(storage.projection)::
268 corners
269 + corner)
270 * transformed.mem(
271 ddc::DiscreteElement<Target>(source));
272 }
273 output.mem(element, ddc::DiscreteElement<Target>(target)) = value;
274 }
275 });
