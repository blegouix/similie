46 """Caller-supplied caloric closure, with no particular equation of state.
47
48 cv=1/T_e. cp=(dh/dT)_p is evaluated as a ratio of Jacobians in (rho,e),
49 with h=e+p/rho. c^2=p_rho+p*p_e/rho^2 follows from de=p/rho^2 drho
50 along an isentrope. The closure must describe one smooth phase with an
51 invertible local (p,T) map, positive heat capacities and sound speed.
52 """

72 def derivatives(self):
73 rho, energy = self.density, self.specific_internal_energy
74 p, temperature = sympify(self.pressure), sympify(self.temperature)
75 p_rho, p_e = diff(p, rho), diff(p, energy)
76 t_rho, t_e = diff(temperature, rho), diff(temperature, energy)
77 enthalpy = energy + p / rho
78 jacobian = simplify(t_e * p_rho - t_rho * p_e)
79 if simplify(t_e) == 0 or jacobian == 0:
80 raise ValueError("Degenerate thermodynamic closure: cv or cp is undefined")
81 return {
82 "pressure_density_derivative": p_rho,
83 "pressure_energy_derivative": p_e,
84 "temperature_density_derivative": t_rho,
85 "temperature_energy_derivative": t_e,
86 "heat_capacity_at_constant_volume": simplify(1 / t_e),
87 "heat_capacity_at_constant_pressure": simplify(
88 (diff(enthalpy, energy) * p_rho - diff(enthalpy, rho) * p_e) / jacobian
89 ),
90 "sound_speed_squared": simplify(p_rho + p * p_e / rho**2),
91 }

93 def integrability_residual(self):
94 """Exactness residual for ds=de/T-p drho/(T rho^2).
95
96 Zero is required for thermodynamic consistency. It is exposed rather
97 than forcing SymPy to decide identities for every user-defined function.
98 """
99 rho, energy = self.density, self.specific_internal_energy
100 return simplify(
101 diff(1 / self.temperature, rho)
102 - diff(-self.pressure / (self.temperature * rho**2), energy)
103 )

174class NavierStokesFourier:
175 """Symbolic mass, momentum, total-energy fluxes and explicit time rates.
176
177 Newtonian stress uses Stokes' zero bulk-viscosity assumption in 3D;
178 Fourier conductivity is mu*cp/Pr. No thermodynamic law is fixed here.
179 """
180
181 def __init__(self):
182 self.density = Symbol("rho", positive=True)
183 self.momentum = Matrix(symbols("m0:3"))
184 self.total_energy = Symbol("rho_E")
185 self.velocity = self.momentum / self.density
186 self.specific_internal_energy = (
187 self.total_energy - self.momentum.dot(self.momentum) / (2 * self.density)
188 ) / self.density
189 self.viscosity, self.prandtl, self.cp, self.pressure = symbols("mu Pr cp p")
190 self.velocity_gradient = Matrix(3, 3, lambda a, b: Symbol(f"g{a}{b}"))
191 self.temperature_gradient = Matrix(symbols("dT0:3"))
192 self.stress = self.viscosity * (
193 self.velocity_gradient
194 + self.velocity_gradient.T
195 - Rational(2, 3) * self.velocity_gradient.trace() * Matrix.eye(3)
196 )
197 self.heat_flux = (
198 -self.viscosity * self.cp / self.prandtl * self.temperature_gradient
199 )
200 self.flux_divergence = Matrix(
201 symbols("div_mass div_m0 div_m1 div_m2 div_energy")
202 )
203 self.evolution = -self.flux_divergence
204
205 def fluxes(self, thermodynamics=None, stress=None, heat_flux=None):
206 p, cp = self.pressure, self.cp
207 if thermodynamics is not None:
208 replacement = {
209 thermodynamics.density: self.density,
210 thermodynamics.specific_internal_energy: self.specific_internal_energy,
211 }
212 p = sympify(thermodynamics.pressure).xreplace(replacement)
213 cp = thermodynamics.derivatives()[
214 "heat_capacity_at_constant_pressure"
215 ].xreplace(replacement)
216 tau = self.stress if stress is None else stress
217 q = self.heat_flux.subs(self.cp, cp) if heat_flux is None else heat_flux
218 flux = Matrix.zeros(5, 3)
219 for normal in range(3):
220 flux[0, normal] = self.momentum[normal]
221 for axis in range(3):
222 flux[axis + 1, normal] = (
223 self.momentum[axis] * self.velocity[normal]
224 + (p if axis == normal else 0)
225 - tau[axis, normal]
226 )
227 flux[4, normal] = (
228 (self.total_energy + p) * self.velocity[normal]
229 - sum(tau[axis, normal] * self.velocity[axis] for axis in range(3))
230 + q[normal]
231 )
232 return flux
