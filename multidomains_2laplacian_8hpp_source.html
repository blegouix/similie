21template <class Direction, class Domains, class Diffusivity>
22auto make_laplacian_flux_domains(
23 Domains const& domains,
24 Diffusivity diffusivity,
25 double jump_scale = 1.0)
26{
27 return domains.transform_data([&](std::size_t id, auto const& data) {
28 double const coefficient = diffusivity(data.physics);
29 if (!(coefficient > 0.0) || !std::isfinite(coefficient))
30 throw std::invalid_argument("scalar diffusivity must be finite and positive");
31 auto const field = domains.field(id);
32 auto const flux = sil::exterior::make_dual_hodge_cochain<Direction>(
33 sil::tensor::make_tensor_with_extrapolation_rules(
34 field,
35 domains.extrapolation_rules(id, jump_scale)),
36 data.geometry,
37 domains.template extrapolation_rules<sil::multidomains::FieldRole::Geometry>(id),
38 domains.template boundary_rules<sil::multidomains::FieldRole::Flux>(id),
39 coefficient);
40 return sil::multidomains::domain_data(data.field, data.geometry, data.physics, flux);
41 });
42}

45template <std::size_t Axis, bool Upper, std::size_t Capacity, class Field, class Primal>
46void prescribe_face(
47 Kokkos::DefaultExecutionSpace const& exec,
48 RowAssembly<Capacity> system,
49 Field const& field,
50 Primal const& primal)
51{
52 if (sil::tensor::detail::is_dirichlet(primal.template boundary_rule<Axis, Upper>())) {
53 auto nodes = field.non_indices_domain();
54 auto front = nodes.front();
55 auto extent = nodes.extents();
56 ddc::detail::array(front)[Axis]
57 = ddc::detail::array(Upper ? nodes.back() : nodes.front())[Axis];
58 ddc::detail::array(extent)[Axis] = 1;
59 // The final endpoint of a physical face can belong to a donor domain.
60 [&]<std::size_t... Tangent>(std::index_sequence<Tangent...>) {
61 (
62 [&] {
63 if constexpr (Tangent != Axis)
64 if (sil::tensor::detail::is_interface(
65 primal.template boundary_rule<Tangent, true>())
66 && !sil::tensor::detail::shared_boundary(
67 primal.template boundary_rule<Tangent, true>()))
68 ++ddc::detail::array(extent)[Tangent];
69 }(),
70 ...);
71 }(std::make_index_sequence<
72 ddc::type_seq_size_v<ddc::to_type_seq_t<typename Field::non_indices_domain_t>>> {});
73 nodes = typename Field::non_indices_domain_t(front, extent);
74 ddc::parallel_for_each(
75 "prescribe_dec_face",
76 exec,
77 nodes,
78 KOKKOS_LAMBDA(typename Field::non_indices_domain_t::discrete_element_type elem) {
79 std::array<std::size_t, Capacity> indices {};
80 auto weights = similie::solvers::detail::make_row_stencil<Capacity>();
81 std::size_t count = 0;
82 double constant = 0.0;
83 detail::visit_rule_basis(
84 primal,
85 field,
86 elem,
87 ddc::DiscreteElement<typename Field::component_type>(0),
88 1.0,
89 [&](auto const& donor, auto sample, auto component, double weight) {
90 if (!similie::solvers::detail::add_stencil_coefficient(
91 weights,
92 indices.data(),
93 count,
94 donor.global_index(sample, component),
95 weight))
96 system.fail(2);
97 },
98 [&](double value) { constant += value; });
99 std::size_t nonzero = 0;
100 std::size_t selected = 0;
101 for (std::size_t i = 0; i < count; ++i)
102 if (weights.mem(
103 ddc::DiscreteElement<similie::solvers::detail::StencilColumn>(
104 i))
105 != 0.0) {
106 ++nonzero;
107 selected = i;
108 }
109 auto exterior = elem;
110 if constexpr (Upper)
111 ++ddc::detail::array(exterior)[Axis];
112 else
113 --ddc::detail::array(exterior)[Axis];
114 auto const& face = primal.template boundary_rule<Axis, Upper>();
115 auto const& boundary = [&]() -> auto const& {
116 if constexpr (requires { face.boundary; })
117 return face.boundary;
118 else
119 return face;
120 }();
121 double const value = sil::tensor::detail::extrapolation_source_value(
122 boundary,
123 field,
124 exterior,
125 ddc::DiscreteElement<typename Field::component_type>(0));
126 if (nonzero != 1)
127 system.fail(8);
128 else
129 system.prescribe(
130 indices[selected],
131 (value - constant)
132 / weights.mem(
133 ddc::DiscreteElement<
134 similie::solvers::detail::StencilColumn>(
135 selected)));
136 });
137 }
138}

140template <std::size_t Capacity, class Field, class Primal, std::size_t... Axis>
141void prescribe_faces(
142 Kokkos::DefaultExecutionSpace const& exec,
143 RowAssembly<Capacity> system,
144 Field const& field,
145 Primal const& primal,
146 std::index_sequence<Axis...>)
147{
148 (prescribe_face<Axis, false, Capacity>(exec, system, field, primal), ...);
149 (prescribe_face<Axis, true, Capacity>(exec, system, field, primal), ...);

198 assemble_differential_rows<Matrix, StencilCoordinates::Relative>(

200 system,

202 flux,
203 op,
204 rules,

206 prescribe_faces<Capacity>(

208 system,
209 field,
210 domains.extrapolation_rules(id, jump_scale),
211 std::make_index_sequence<Direction::size()> {});
