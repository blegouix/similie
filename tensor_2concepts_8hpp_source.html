49// All indices expose the same metadata and access operation. Explicit storage
50// additionally provides a direct map between access IDs and memory IDs.
51template <class Index>
52concept TensorIndexConcept = requires(
53 detail::IndexTensorConcept tensor,
54 ddc::DiscreteElement<Index> element,
55 detail::IndexAccessConcept access) {
56 requires Index::is_tensor_index;
57 typename std::bool_constant<Index::is_explicitely_stored_tensor>;
58 typename Index::subindices_domain_t;
59 { Index::subindices_domain() } -> std::same_as<typename Index::subindices_domain_t>;
60 { Index::rank() } -> std::same_as<std::size_t>;
61 { Index::size() } -> std::same_as<std::size_t>;
62 { Index::mem_size() } -> std::same_as<std::size_t>;
63 { Index::access_size() } -> std::same_as<std::size_t>;
64 typename std::integral_constant<std::size_t, Index::rank()>;
65 typename std::integral_constant<std::size_t, Index::size()>;
66 typename std::integral_constant<std::size_t, Index::mem_size()>;
67 typename std::integral_constant<std::size_t, Index::access_size()>;
68 {
69 Index::template process_access<
70 detail::IndexTensorConcept,
71 ddc::DiscreteElement<Index>,
72 Index>(access, tensor, element)
73 } -> std::same_as<double>;
74 requires(!Index::is_explicitely_stored_tensor || requires(std::size_t id) {
75 { Index::access_id_to_mem_id(id) } -> std::same_as<std::size_t>;
76 {
77 Index::mem_id_to_canonical_natural_ids(id)
78 } -> std::same_as<std::array<std::size_t, Index::rank()>>;
79 // Natural indices take one ID; composite indices take subindex IDs.
80 requires (requires(std::array<std::size_t, Index::subindices_domain_t::rank()> const& ids) {
81 { Index::mem_id(ids) } -> std::same_as<std::size_t>;
82 { Index::access_id(ids) } -> std::same_as<std::size_t>;
83 } || requires {
84 requires Index::is_tensor_natural_index;
85 { Index::mem_id(id) } -> std::same_as<std::size_t>;
86 { Index::access_id(id) } -> std::same_as<std::size_t>;
87 });
88 });
89};
