24template <class Result, class Contribution>
25KOKKOS_FUNCTION void accumulate_extrapolation(
26 Result& result,
27 Contribution const& contribution,
28 double weight,
29 std::size_t source)
30{
31 if constexpr (ddc::type_seq_contains_v<
32 ddc::TypeSeq<ExtrapolationSource>,
33 ddc::to_type_seq_t<typename Result::discrete_domain_type>>) {
34 using Spatial = misc::convert_type_seq_to_t<
35 ddc::DiscreteDomain,
36 ddc::type_seq_remove_t<
37 ddc::to_type_seq_t<typename Result::discrete_domain_type>,
38 ddc::TypeSeq<ExtrapolationSource>>>;
39 ddc::device_for_each(contribution.domain(), [&](auto entry) {
40 result
41 .mem(typename Spatial::discrete_element_type(entry),
42 ddc::DiscreteElement<ExtrapolationSource>(
43 entry.template uid<ExtrapolationSource>() == 0 ? 0 : source))
44 += weight * contribution.mem(entry);
45 });
46 } else
47 ddc::device_for_each(result.domain(), [&](auto component) {
48 result.mem(component) += weight * contribution.mem(component);
49 });
50}

97 template <class Domain, class Element, class Result, class Evaluate>
98 KOKKOS_FUNCTION void accumulate(
99 Domain const& domain,
100 Element element,
101 Result& result,
102 double& total,
103 Evaluate const& evaluate) const
104 {
105 using DDim = ddc::type_seq_element_t<Dimension, ddc::to_type_seq_t<Domain>>;
106 std::ptrdiff_t const sampled = static_cast<std::ptrdiff_t>(element.template uid<DDim>());
107 std::ptrdiff_t const front
108 = static_cast<std::ptrdiff_t>(domain.front().template uid<DDim>());
109 std::ptrdiff_t const back = static_cast<std::ptrdiff_t>(domain.back().template uid<DDim>());
110 double const distance = sampled < front ? double(front - sampled)
111 : sampled > back ? double(sampled - back)
112 : 0.0;
113 if (distance != 0.0) {
114 if (sampled < front)
115 accumulate_extrapolation(
116 result,
117 evaluate(m_rules.first, element),
118 distance,
119 1 + 2 * Dimension);
120 else
121 accumulate_extrapolation(
122 result,
123 evaluate(m_rules.second, element),
124 distance,
125 2 + 2 * Dimension);
126 total += distance;
127 }
128 m_other_rules.accumulate(domain, element, result, total, evaluate);
129 }
130};

194 template <class Domain, class Element>
195 KOKKOS_FUNCTION auto value(Domain const& domain, Element sampled_element) const
196 {
197 static_assert(sizeof...(RulePairs) == ddc::type_seq_size_v<ddc::to_type_seq_t<Domain>>);
198 misc::convert_type_seq_to_t<misc::SignedDiscreteElement, ddc::to_type_seq_t<Domain>> const
199 element(sampled_element);
200 auto result
201 = detail::make_extrapolation_basis<1 + 2 * sizeof...(RulePairs)>(domain, element);
202 if (misc::domain_contains(domain, element)) {
203 result
204 .mem(element.storage_element(),
205 ddc::DiscreteElement<detail::ExtrapolationSource>(0))
206 = 1.0;
207 return result;
208 }
209 double total = 0.0;
210 m_rules.accumulate(domain, element, result, total, [&](auto const& rule, auto sample) {
211 return rule.value(domain, sample);
212 });
213 result *= 1.0 / total;
