189 KOKKOS_FUNCTION element_type integrate() noexcept
190 {
191 element_type out = 0;
192 for (auto i = begin(); i < end(); ++i) {
193 if constexpr (misc::Specialization<chain_type, Chain>) {
194 out += (i->negative() ? -1 : 1) * i->value();
195 } else if constexpr (misc::Specialization<chain_type, LocalChain>) {
196 out += (i->negative() ? -1 : 1) * i->value();
197 }
198 }
199 return out;
200 }

207 out += (i->negative() ? -1 : 1) * i->value();
208 } else if constexpr (misc::Specialization<chain_type, LocalChain>) {
