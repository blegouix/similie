84 [[nodiscard]] KOKKOS_FUNCTION coupling::ElectromagneticExchange coupling_sources(
85 State const& state,
86 Vector const& total_current,
87 Vector const& velocity) const
88 {
89 coupling::ElectromagneticCouplingState shared {};
90 shared.current = total_current;
91 if constexpr (has_electric) {
92 shared.charge_density = state.electromagnetic.charge_density;
93 shared.electric_field
94 = this->template get<1>().electric_field(state.electromagnetic.displacement);
95 } else {
96 shared.electric_field = this->template get<1>().ohm_electric_field(
97 velocity,
98 state.electromagnetic.induction,
99 total_current);
100 }
101 if constexpr (has_magnetic)
102 shared.induction = state.electromagnetic.induction;
103 coupling::ElectromagneticExchange result = this->template get<2>()(shared);
104 result += this->template get<3>()(shared);
105 return result;
106 }

176 template <class Fluid = FluidPhysics>
177 [[nodiscard]] KOKKOS_FUNCTION typename Fluid::Flux flux(
178 State const& state,
179 typename Fluid::VelocityGradient const& velocity_gradient,
180 Vector const& temperature_gradient,
181 int normal,
182 Vector const& total_current) const
183 {
184 KOKKOS_ASSERT(normal >= 0 && normal < 3);
185 typename Fluid::State const hydro = fluid_state(state);
186 typename Fluid::PrimitiveVariables const primitive
187 = this->template get<0>().primitives(hydro);
188 typename Fluid::Flux result
189 = this->template get<0>()
190 .flux(hydro, primitive, velocity_gradient, temperature_gradient, normal);
191 if constexpr (Energy == EnergyConvention::FluidAndField) {
192 static_assert(
193 has_magnetic,
194 "Electric quasistatics uses fluid energy with J.E exchange");
195 if constexpr (has_electric) {
196 Vector const poynting = this->template get<1>().poynting_flux(
197 state.electromagnetic.displacement,
198 state.electromagnetic.induction);
199 result.total_energy += poynting[normal];
200 } else {
201 Vector const electric = this->template get<1>().ohm_electric_field(
202 primitive.velocity,
203 state.electromagnetic.induction,
204 total_current);
205 Vector const poynting
206 = this->template get<1>()
207 .poynting_flux(electric, state.electromagnetic.induction);
208 result.total_energy += poynting[normal];
209 }
210 }
211 return result;
212 }

225 template <class FluidFlux>
226 [[nodiscard]] KOKKOS_FUNCTION State operator()(
227 State const& state,
228 FluidFlux const& fluid_flux_divergence,
229 double current_divergence,
230 Vector const& curl_magnetic_field,
231 Vector const& curl_electric_field,
232 Vector const& total_current) const
233 {
234 typename FluidPhysics::State rate = this->template get<0>()(fluid_flux_divergence);
235 coupling::ElectromagneticExchange const exchange = coupling_sources(
236 state,
237 total_current,
238 this->template get<0>().primitives(fluid_state(state)).velocity);
239 for (int a = 0; a < 3; ++a)
240 rate.momentum[a] += exchange.momentum_source[a];
241 if constexpr (Energy == EnergyConvention::Fluid)
242 rate.total_energy += exchange.fluid_energy_source;
243 return {rate,
244 field_rate(
245 current_divergence,
246 curl_magnetic_field,
247 curl_electric_field,
248 total_current)};
249 }

259 [[nodiscard]] KOKKOS_FUNCTION Vector
260 transport(State const& state, Vector const& velocity, Vector const& total_current) const
261 {
262 Vector result = this->template get<0>().transport(velocity, state.fluid.vorticity);
263 coupling::ElectromagneticExchange const exchange
264 = coupling_sources(state, total_current, velocity);
265 for (int a = 0; a < 3; ++a)
266 result[a] += exchange.momentum_source[a] / this->template get<0>().density;
267 return result;
