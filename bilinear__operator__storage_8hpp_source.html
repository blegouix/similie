82 BilinearSpatialOperatorStorage result;
83 result.m_domain = first.non_indices_domain();
84 result.m_first_domain = first.non_indices_domain();
85 result.m_second_domain = second.non_indices_domain();
86 // Equal spatial domains keep sampling and cached boundary rules unambiguous.
87 if (result.m_first_domain != result.m_second_domain)
88 throw std::invalid_argument("Bilinear staging requires equal input spatial domains");
89 result.m_weights = decltype(m_weights)(
90 Kokkos::view_alloc(Kokkos::WithoutInitializing, "staged_bilinear_stencils"),
91 result.m_domain.size());
92 auto const weights = result.m_weights;
93 Domain const domain = result.m_domain;
94 std::tuple<Arguments...> const cached_arguments(arguments...);
95 Kokkos::View<int, Memory> incomplete("staged_bilinear_incomplete");
96 Kokkos::parallel_for(
97 "similie_cache_bilinear_operator_stencils",
98 Kokkos::RangePolicy<Exec>(exec, 0, domain.size()),
99 KOKKOS_LAMBDA(std::size_t node) {
100 Coefficients const stencil = evaluate_value(
101 geometry,
102 first,
103 second,
104 detail::stencil_element(domain, node),
105 cached_arguments,
106 std::index_sequence_for<Arguments...>());
107 if (!stencil.complete)
108 Kokkos::atomic_max(&incomplete(), 1);
109 for (std::size_t entry = 0; entry < stencil.weights.size(); ++entry)
110 weights(node, entry) = stencil.weights[entry];
111 });
112 int failed;
113 Kokkos::deep_copy(exec, failed, incomplete);
114 exec.fence();
115 if (failed)

130 template <class Output, class FirstTensor, class SecondTensor>
131 requires(
132 FirstRadius == 0 && SecondRadius == 0
133 && std::same_as<typename FirstTensor::indices_domain_t, ddc::DiscreteDomain<First>>
134 && std::same_as<
135 typename SecondTensor::indices_domain_t,
136 ddc::DiscreteDomain<Second>>
137 && std::same_as<typename Output::indices_domain_t, ddc::DiscreteDomain<Target>>)
138 KOKKOS_FUNCTION void apply_pointwise(
139 Output const& output,
140 FirstTensor const& first,
141 SecondTensor const& second,
142 typename Domain::discrete_element_type element) const
143 {
144 std::size_t const node = detail::stencil_node(m_domain, element);
145 for (std::size_t target = 0; target < Target::mem_size(); ++target) {
146 double sum = 0.0;
147 for (std::size_t a = 0; a < First::mem_size(); ++a)
148 for (std::size_t b = 0; b < Second::mem_size(); ++b)
149 sum += m_weights(
150 node,
151 (target * First::mem_size() + a) * Second::mem_size() + b)
152 * first.mem(ddc::DiscreteElement<First>(a))
153 * second.mem(ddc::DiscreteElement<Second>(b));
154 output.mem(ddc::DiscreteElement<Target>(target)) = sum;
155 }
156 }

158 template <class Output, class FirstField, class SecondField>
159 requires(
160 std::same_as<typename FirstField::indices_domain_t, ddc::DiscreteDomain<First>>
161 && std::same_as<typename SecondField::indices_domain_t, ddc::DiscreteDomain<Second>>
162 && std::same_as<typename Output::indices_domain_t, ddc::DiscreteDomain<Target>>)
163 Output apply(
164 Exec const& exec,
165 Output const& output,
166 FirstField const& first,
167 SecondField const& second) const
168 {
169 if (first.non_indices_domain() != m_first_domain
170 || second.non_indices_domain() != m_second_domain)
171 throw std::invalid_argument("Staged bilinear operator input domain changed");
172 Domain const out_domain = output.non_indices_domain();
173 if (!out_domain.empty()
174 && (!misc::domain_contains(m_domain, out_domain.front())
175 || !misc::domain_contains(m_domain, out_domain.back())))
176 throw std::invalid_argument("Staged bilinear operator output exceeds cached domain");
177 auto const weights = m_weights;
178 Domain const domain = m_domain;
179 Kokkos::parallel_for(
180 "similie_apply_staged_bilinear_stencil",
181 Kokkos::RangePolicy<Exec>(exec, 0, out_domain.size() * Target::mem_size()),
182 KOKKOS_LAMBDA(std::size_t row) {
183 auto const element
184 = detail::stencil_element(out_domain, row / Target::mem_size());
185 std::size_t const node = detail::stencil_node(domain, element),
186 target = row % Target::mem_size();
187 double sum = 0.0;
188 for (std::size_t a = 0; a < Coefficients::first_size; ++a)
189 for (std::size_t b = 0; b < Coefficients::second_size; ++b) {
190 double const coefficient = weights(
191 node,
192 (target * Coefficients::first_size + a)
193 * Coefficients::second_size
194 + b);
195 if (coefficient != 0.0)
196 sum += coefficient
197 * first
198 .mem(detail::stencil_source<Index, FirstRadius>(
199 domain,
200 element,
201 a / First::mem_size()),
202 ddc::DiscreteElement<First>(
203 a % First::mem_size()))
204 * second
205 .mem(detail::stencil_source<Index, SecondRadius>(
206 domain,
207 element,
208 b / Second::mem_size()),
209 ddc::DiscreteElement<Second>(
210 b % Second::mem_size()));
211 }
212 output.mem(element, ddc::DiscreteElement<Target>(target)) = sum;
213 });
214 return output;
215 }
