33 Matrix m_matrix {};

36 template <class StressLaw>
37 ElasticMaterialHodge2D(Positions const& positions, StressLaw stress_law)
38 {
39 double const signed_area
40 = 0.5
41 * (positions[0][0] * positions[1][1] - positions[0][1] * positions[1][0]
42 + positions[1][0] * positions[3][1] - positions[1][1] * positions[3][0]
43 + positions[3][0] * positions[2][1] - positions[3][1] * positions[2][0]
44 + positions[2][0] * positions[0][1] - positions[2][1] * positions[0][0]);
45 if (!(signed_area > 0.0))
46 throw std::runtime_error("elasticity cell has nonpositive area");
47 for (int corner = 0; corner < 4; ++corner) {
48 int const next_x = corner ^ 1;
49 int const next_y = corner ^ 2;
50 double const dx_x
51 = (corner & 1 ? -1.0 : 1.0) * (positions[next_x][0] - positions[corner][0]);
52 double const dx_y
53 = (corner & 1 ? -1.0 : 1.0) * (positions[next_x][1] - positions[corner][1]);
54 double const dy_x
55 = (corner & 2 ? -1.0 : 1.0) * (positions[next_y][0] - positions[corner][0]);
56 double const dy_y
57 = (corner & 2 ? -1.0 : 1.0) * (positions[next_y][1] - positions[corner][1]);
58 if (!(dx_x * dy_y - dx_y * dy_x > 0.0))
59 throw std::runtime_error("elasticity cell has folded geometry");
60 }
61
62 constexpr std::array<std::array<int, 2>, 4> endpoints {{{0, 1}, {2, 3}, {0, 2}, {1, 3}}};
63 std::array<std::array<double, 4>, 8> affine {};
64 std::array<std::array<double, 4>, 8> physical_force {};
65 std::array<double, 2> centre {};
66 for (int vertex = 0; vertex < 4; ++vertex)
67 for (int axis = 0; axis < 2; ++axis)
68 centre[axis] += 0.25 * positions[vertex][axis];
69 for (int edge = 0; edge < 4; ++edge) {
70 int const tail = endpoints[edge][0];
71 int const head = endpoints[edge][1];
72 double const dx = positions[head][0] - positions[tail][0];
73 double const dy = positions[head][1] - positions[tail][1];
74 affine[2 * edge] = {dx, dy, 0.0, 0.0};
75 affine[2 * edge + 1] = {0.0, 0.0, dx, dy};
76 double const mx = 0.5 * (positions[tail][0] + positions[head][0]);
77 double const my = 0.5 * (positions[tail][1] + positions[head][1]);
78 double const vx = (edge == 0 || edge == 2 ? 1.0 : -1.0) * (centre[0] - mx);
79 double const vy = (edge == 0 || edge == 2 ? 1.0 : -1.0) * (centre[1] - my);
80 double const nx = edge < 2 ? vy : -vy;
81 double const ny = edge < 2 ? -vx : vx;
82 for (int gradient = 0; gradient < 4; ++gradient) {
83 Strain2D const strain = DisplacementToStrain::from_gradient(
84 gradient == 0 ? 1.0 : 0.0,
85 gradient == 3 ? 1.0 : 0.0,
86 gradient == 1 ? 1.0 : 0.0,
87 gradient == 2 ? 1.0 : 0.0);
88 CauchyStress2D const stress = stress_law(strain);
89 physical_force[2 * edge][gradient] = stress.xx * nx + stress.xy * ny;
90 physical_force[2 * edge + 1][gradient] = stress.xy * nx + stress.yy * ny;
91 }
92 }
93
94 std::array<std::array<double, 8>, 4> augmented {};
95 for (int i = 0; i < 4; ++i) {
96 for (int j = 0; j < 4; ++j)
97 for (int edge_component = 0; edge_component < 8; ++edge_component)
98 augmented[i][j] += affine[edge_component][i] * affine[edge_component][j];
99 augmented[i][4 + i] = 1.0;
100 }
101 for (int pivot = 0; pivot < 4; ++pivot) {
102 int best = pivot;
103 for (int row = pivot + 1; row < 4; ++row)
104 if (std::abs(augmented[row][pivot]) > std::abs(augmented[best][pivot]))
105 best = row;
106 if (!(std::abs(augmented[best][pivot]) > 1e-25 * signed_area))
107 throw std::runtime_error("degenerate elasticity cell");
108 std::swap(augmented[best], augmented[pivot]);
109 double const divisor = augmented[pivot][pivot];
110 for (int column = 0; column < 8; ++column)
111 augmented[pivot][column] /= divisor;
112 for (int row = 0; row < 4; ++row) {
113 if (row == pivot)
114 continue;
115 double const factor = augmented[row][pivot];
116 for (int column = 0; column < 8; ++column)
117 augmented[row][column] -= factor * augmented[pivot][column];
118 }
119 }
120 for (int i = 0; i < 4; ++i)
121 for (int edge_component = 0; edge_component < 8; ++edge_component)
122 for (int j = 0; j < 4; ++j)
123 m_affine_recovery[i][edge_component]
124 += augmented[i][4 + j] * affine[edge_component][j];
125
126 std::array<std::array<double, 4>, 4> affine_energy {};
127 for (int i = 0; i < 4; ++i)
128 for (int j = 0; j < 4; ++j)
129 for (int component = 0; component < 8; ++component)
130 affine_energy[i][j] += affine[component][i] * physical_force[component][j];
131 // Symmetric extension of H E = B. It supplies exact affine tractions,
132 // while the complementary projector controls unresolved edge modes.
133 double scale = 0.0;
134 for (int i = 0; i < 8; ++i)
135 for (int j = 0; j < 8; ++j) {
136 for (int k = 0; k < 4; ++k) {
137 m_matrix[i][j] += physical_force[i][k] * m_affine_recovery[k][j]
138 + m_affine_recovery[k][i] * physical_force[j][k];
139 for (int l = 0; l < 4; ++l)
140 m_matrix[i][j] -= m_affine_recovery[k][i] * affine_energy[k][l]
141 * m_affine_recovery[l][j];
142 }
143 scale = std::max(scale, std::abs(m_matrix[i][j]));
144 }
145 Matrix complement {};
146 for (int i = 0; i < 8; ++i)
147 for (int j = 0; j < 8; ++j) {
148 complement[i][j] = i == j ? 1.0 : 0.0;
149 for (int k = 0; k < 4; ++k)
150 complement[i][j] -= affine[i][k] * m_affine_recovery[k][j];
151 }
152 double coupling_norm_squared = 0.0;
153 for (int i = 0; i < 8; ++i)
154 for (int gradient = 0; gradient < 4; ++gradient) {
155 double coupling = 0.0;
156 for (int j = 0; j < 8; ++j)
157 coupling += complement[i][j] * physical_force[j][gradient];
158 coupling_norm_squared += coupling * coupling;
159 }
160 double const shear_modulus = 0.5 * stress_law(Strain2D {.xy = 1.0}).xy;
161 if (!(shear_modulus > 0.0))
162 throw std::runtime_error("elasticity material has nonpositive shear modulus");
163 // The second bound dominates the affine/complement coupling in the
164 // Schur complement; the first controls the otherwise free modes.
165 double const stabilization = std::
166 max(0.25 * scale,
167 1.01 * coupling_norm_squared / (2.0 * shear_modulus * signed_area));
168 for (int i = 0; i < 8; ++i)
169 for (int j = 0; j < 8; ++j)
170 m_matrix[i][j] += stabilization * complement[i][j];
171 }

178 [[nodiscard]] std::array<double, 4> recover_gradient(Vector const& differences) const
179 {
180 std::array<double, 4> gradient {};
181 for (int i = 0; i < 4; ++i)
182 for (int j = 0; j < 8; ++j)
183 gradient[i] += m_affine_recovery[i][j] * differences[j];
184 return gradient;

187 template <class SpatialX, class SpatialY>
188 [[nodiscard]] static Vector edge_differences(std::array<std::array<double, 2>, 4> const& nodal)
189 {
190 Vector differences {};
191 constexpr std::array<std::array<std::size_t, 2>, 4> edges {
192 {{0, 0}, {0, 2}, {1, 0}, {1, 1}}};
193 for (int edge = 0; edge < 4; ++edge) {
194 std::array<double, 2> const difference = sil::exterior::
195 CovariantDerivative<SpatialX, SpatialY>::template cochain_value<0, 2>(
196 {edges[edge][0]},
197 edges[edge][1],
198 [&](std::size_t, std::size_t vertex) { return nodal[vertex]; });
199 differences[2 * edge] = difference[0];
200 differences[2 * edge + 1] = difference[1];
201 }
202 return differences;
203 }
